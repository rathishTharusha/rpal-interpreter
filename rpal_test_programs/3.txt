let rec fib n =
    n eq 1 -> 0 |
    n eq 2 -> 1 |
    fib(n-1) + fib(n-2)
in
let rec get_fibs_in_range (n, start_val, end_val) = 
    let f = fib n in
    f > end_val -> nil |
    f >= start_val -> (f, get_fibs_in_range(n+1, start_val, end_val)) |
    get_fibs_in_range(n+1, start_val, end_val)
in
let start_range = 5 in
let end_range = 50 in
Print (get_fibs_in_range(1, start_range, end_range))
